Practical 5: MAnaging undo Data

a)Monitor and administrator undo
->select name, value from v$parameter where name like 'undo%';

select 
a.tablespace_name,
a.bytes/1024/1024 as size_mb,
b.bytes/1024/1024 as used_mb,
(a.bytes-NVL(b.bytes,0))/1024/1024 as free_mb
from
dba_data_files a,
(select sum(bytes) bytes, file_id
from dba_undo_extents
where status='ACTIVE'
Group by file_id) b
where a.file_id=b.file_id(+)
and a.tablespace_name in (select value from v$parameter where name='undo_tablespace');

select
s.sid,
s.serial#,
t.used_urec as "Undo Records",
t.used_ublk as "Undo Blocks",
s.username
from
v$transaction t,
v$session s
where
t.ses_addr=s.saddr;
